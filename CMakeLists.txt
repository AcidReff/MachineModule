cmake_minimum_required(VERSION 3.22)
project(MachinedrumModulePC VERSION 0.4.4 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(MSVC)
  add_compile_definitions(_USE_MATH_DEFINES NOMINMAX _CRT_SECURE_NO_WARNINGS)
  add_compile_options(/utf-8 /bigobj /permissive- /constexpr:steps10000000)
endif()
set(ASMJIT_STATIC TRUE)
set(ASMJIT_NO_INSTALL TRUE)
add_subdirectory(libs/dsp56300/source/asmjit EXCLUDE_FROM_ALL)
add_subdirectory(libs/dsp56300-jit/source/dsp56kBase EXCLUDE_FROM_ALL)
add_subdirectory(libs/dsp56300-jit/source/dsp56kEmu EXCLUDE_FROM_ALL)
if(MSVC)
  target_compile_options(dsp56kEmu PRIVATE /MP4 /wd4828)
endif()
if(WIN32 OR (UNIX AND NOT APPLE))
  add_subdirectory(libs/dsp56300-jit/source/vtuneSdk EXCLUDE_FROM_ALL)
endif()
add_subdirectory(libs/gearmulator-md-mm/source/mc68k EXCLUDE_FROM_ALL)
add_library(mdcore STATIC engine/HostModel.cpp engine/MachineRunner.cpp engine/Mixer.cpp
  engine/ParallelVoiceEngine.cpp engine/TrackFx.cpp engine/VoiceEngine.cpp tools/mdfw/Firmware.cpp tools/mdmix/MixerRef.cpp)
target_include_directories(mdcore PUBLIC engine tools/mdfw PRIVATE libs/gearmulator-md-mm/source libs/gearmulator-md-mm/source/mc68k)
target_link_libraries(mdcore PUBLIC dsp56kEmu 68kEmu)
add_executable(md-smoke Source/Smoke.cpp)
add_executable(md-jit-smoke Source/JitSmoke.cpp)
target_link_libraries(md-jit-smoke PRIVATE mdcore)
add_executable(md-render-benchmark Source/RenderBenchmark.cpp)
target_link_libraries(md-render-benchmark PRIVATE mdcore)
add_executable(md-lfo-smoke Source/LfoSmoke.cpp)
target_link_libraries(md-lfo-smoke PRIVATE mdcore)
add_executable(md-x13-smoke Source/X13Smoke.cpp)
target_link_libraries(md-x13-smoke PRIVATE mdcore)
add_executable(md-firmware-probe Source/FirmwareProbe.cpp)
target_link_libraries(md-firmware-probe PRIVATE mdcore)
target_include_directories(md-firmware-probe PRIVATE libs/gearmulator-md-mm/source)
target_link_libraries(md-smoke PRIVATE mdcore)
add_executable(md-master-inspect Source/MasterInspect.cpp)
add_executable(md-master-probe Source/MasterProbe.cpp)
add_executable(md-cpu-benchmark Source/CpuBenchmark.cpp)
target_link_libraries(md-cpu-benchmark PRIVATE mdcore)
target_link_libraries(md-master-probe PRIVATE mdcore)
target_include_directories(md-master-inspect PRIVATE libs/gearmulator-md-mm/source/mc68k)
target_link_libraries(md-master-inspect PRIVATE mdcore)
set(JUCE_DIR "C:/JUCE" CACHE PATH "JUCE source directory")
add_subdirectory("${JUCE_DIR}" "${CMAKE_BINARY_DIR}/JUCE" EXCLUDE_FROM_ALL)
juce_add_plugin(MachinedrumModule
  PRODUCT_NAME "MachineModule" COMPANY_NAME "MachinedrumModulePC"
  BUNDLE_ID "org.machinedrummodule.pc" PLUGIN_MANUFACTURER_CODE Mdpc PLUGIN_CODE Mdp1
  FORMATS VST3 Standalone IS_SYNTH TRUE NEEDS_MIDI_INPUT TRUE NEEDS_MIDI_OUTPUT FALSE
  COPY_PLUGIN_AFTER_BUILD FALSE EDITOR_WANTS_KEYBOARD_FOCUS FALSE)
if(WIN32)
  # JUCE 8.0.8's resource command omits its metadata input dependency.
  # Regenerate Windows name/version resources when product metadata changes.
  get_target_property(md_generated MachinedrumModule JUCE_GENERATED_SOURCES_DIRECTORY)
  get_target_property(md_info MachinedrumModule JUCE_INFO_FILE)
  add_custom_command(OUTPUT "${md_generated}/MachinedrumModule_resources.rc" APPEND DEPENDS "${md_info}")
endif()
target_sources(MachinedrumModule PRIVATE Source/Plugin.cpp)
set_target_properties(MachinedrumModule_Standalone PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/dist/Standalone/$<CONFIG>")
target_compile_definitions(MachinedrumModule PUBLIC JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_VST3_CAN_REPLACE_VST2=0)
target_link_libraries(MachinedrumModule PRIVATE mdcore juce::juce_audio_utils PUBLIC juce::juce_recommended_config_flags)
juce_add_console_app(md-plugin-smoke PRODUCT_NAME "md-plugin-smoke")
target_sources(md-plugin-smoke PRIVATE Source/PluginSmoke.cpp)
target_include_directories(md-plugin-smoke PRIVATE "${JUCE_DIR}/modules")
target_compile_definitions(md-plugin-smoke PRIVATE JUCE_GLOBAL_MODULE_SETTINGS_INCLUDED=1 JUCE_STANDALONE_APPLICATION=1 JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0)
target_link_libraries(md-plugin-smoke PRIVATE MachinedrumModule mdcore)

juce_add_console_app(md-midi-smoke PRODUCT_NAME "md-midi-smoke")
target_sources(md-midi-smoke PRIVATE Source/MidiSmoke.cpp)
target_include_directories(md-midi-smoke PRIVATE "${JUCE_DIR}/modules")
target_compile_definitions(md-midi-smoke PRIVATE JUCE_GLOBAL_MODULE_SETTINGS_INCLUDED=1 JUCE_STANDALONE_APPLICATION=1 JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0)
target_link_libraries(md-midi-smoke PRIVATE MachinedrumModule mdcore)

juce_add_console_app(md-processor-benchmark PRODUCT_NAME "md-processor-benchmark")
target_sources(md-processor-benchmark PRIVATE Source/ProcessorBenchmark.cpp)
target_include_directories(md-processor-benchmark PRIVATE "${JUCE_DIR}/modules")
target_compile_definitions(md-processor-benchmark PRIVATE JUCE_GLOBAL_MODULE_SETTINGS_INCLUDED=1 JUCE_STANDALONE_APPLICATION=1 JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0)
target_link_libraries(md-processor-benchmark PRIVATE MachinedrumModule mdcore)

# Optional comparison against a saved Plugin.cpp; never part of the release.
set(MACHINEMODULE_REFERENCE_PLUGIN "" CACHE FILEPATH "Previous Plugin.cpp for routing A/B validation")
if(MACHINEMODULE_REFERENCE_PLUGIN)
  juce_add_console_app(md-processor-reference PRODUCT_NAME "md-processor-reference")
  target_sources(md-processor-reference PRIVATE Source/ProcessorBenchmark.cpp "${MACHINEMODULE_REFERENCE_PLUGIN}")
  target_include_directories(md-processor-reference PRIVATE Source "${JUCE_DIR}/modules")
  target_compile_definitions(md-processor-reference PRIVATE JUCE_GLOBAL_MODULE_SETTINGS_INCLUDED=1 JUCE_STANDALONE_APPLICATION=1 JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0)
  target_link_libraries(md-processor-reference PRIVATE MachinedrumModule mdcore)
endif()
